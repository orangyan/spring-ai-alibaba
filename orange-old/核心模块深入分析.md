```java
public class DashScopeChatModel extends AbstractChatModel {
    
    private final DashScopeApi dashscopeApi;
    private final DashScopeChatOptions defaultOptions;
    
    @Override
    public ChatResponse call(Prompt prompt) {
        // 1. 构建请求
        DashScopeApi.ChatCompletionRequest request = 
            createRequest(prompt, false);
        
        // 2. 调用 API
        ResponseEntity<ChatCompletion> response = 
            this.dashscopeApi.chatCompletionEntity(request);
        
        // 3. 转换响应
        return toChatResponse(response.getBody());
    }
    
    @Override
    public Flux<ChatResponse> stream(Prompt prompt) {
        // 1. 构建流式请求
        DashScopeApi.ChatCompletionRequest request = 
            createRequest(prompt, true);
        
        // 2. 流式调用
        Flux<ChatCompletionChunk> flux = 
            this.dashscopeApi.chatCompletionStream(request);
        
        // 3. 转换流式响应
        return flux
            .map(this::toChatResponse)
            .filter(Objects::nonNull);
    }
    
    // 创建请求
    private ChatCompletionRequest createRequest(
            Prompt prompt, boolean stream) {
        
        // 合并选项
        ChatOptions options = mergeOptions(
            prompt.getOptions(),
            this.defaultOptions
        );
        
        // 转换消息
        List<Message> messages = prompt.getInstructions()
            .stream()
            .map(this::toApiMessage)
            .toList();
        
        // 构建请求
        return ChatCompletionRequest.builder()
            .model(options.getModel())
            .messages(messages)
            .temperature(options.getTemperature())
            .maxTokens(options.getMaxTokens())
            .stream(stream)
            .build();
    }
}
```

### 2. Function Calling 实现

#### 机制流程

```
1. 用户输入
   ↓
2. ChatClient 发送请求 + 函数定义
   ↓
3. 模型识别需要调用函数
   ↓
4. 模型返回函数调用请求
   {
     "function_call": {
       "name": "getWeather",
       "arguments": "{\"city\":\"北京\"}"
     }
   }
   ↓
5. ChatClient 执行本地函数
   result = getWeather("北京")
   ↓
6. 将结果作为消息发回模型
   ↓
7. 模型基于函数结果生成最终回答
```

#### 源码实现

```java
public class FunctionCallingSupport {
    
    public ChatResponse handleFunctionCalling(
            Prompt prompt,
            Map<String, Function<?, ?>> functions) {
        
        // 1. 第一次调用 - 带函数定义
        ChatResponse response = callWithFunctions(prompt, functions);
        
        // 2. 检查是否需要函数调用
        while (hasFunctionCall(response)) {
            // 3. 提取函数调用信息
            FunctionCall functionCall = extractFunctionCall(response);
            
            // 4. 执行函数
            Object result = executeFunction(
                functionCall.getName(),
                functionCall.getArguments(),
                functions
            );
            
            // 5. 构建新的 Prompt (包含函数结果)
            Prompt newPrompt = Prompt.builder()
                .messages(prompt.getInstructions())
                .message(new FunctionMessage(
                    functionCall.getName(),
                    objectMapper.writeValueAsString(result)
                ))
                .build();
            
            // 6. 再次调用模型
            response = callWithFunctions(newPrompt, functions);
        }
        
        return response;
    }
    
    private Object executeFunction(
            String functionName,
            String argumentsJson,
            Map<String, Function<?, ?>> functions) {
        
        Function<?, ?> function = functions.get(functionName);
        if (function == null) {
            throw new IllegalArgumentException(
                "Function not found: " + functionName);
        }
        
        // 反序列化参数
        Object arguments = objectMapper.readValue(
            argumentsJson,
            function.getParameterType()
        );
        
        // 执行函数
        return function.apply(arguments);
    }
}
```

```java
    @Override
    public AdvisedRequest adviseRequest(
            AdvisedRequest request, 
            Map<String, Object> context) {
        
        // 1. 提取用户查询

            query, 
            relevantDocs
        );

        return AdvisedRequest.from(request)
            .withUserText(augmentedPrompt)
            .build();
    }
    
    private String buildAugmentedPrompt(
            String query, 
            List<Document> docs) {
        
        StringBuilder contextBuilder = new StringBuilder();
        contextBuilder.append("根据以下上下文回答问题：\n\n");
        
        // 添加检索到的文档
        for (int i = 0; i < docs.size(); i++) {
            contextBuilder
                .append("文档 ")
                .append(i + 1)
                .append(":\n")
                .append(docs.get(i).getText())
                .append("\n\n");
        }
        
        contextBuilder.append("问题：").append(query);
        
        return contextBuilder.toString();
    }
```

```java
public class RerankEnhancedRetriever implements DocumentRetriever {

    private final RerankModel rerankModel;
    
    @Override
    public List<Document> retrieve(String query) {
        // 1. 初始检索 (召回更多文档)
        List<Document> candidates = vectorStore.similaritySearch(
            SearchRequest.query(query)
                .withTopK(20)  // 召回 20 个候选
        );
        
        // 2. 重排序
        List<RerankResult> rerankResults = rerankModel.rerank(
            query,
            candidates.stream()
                .map(Document::getText)
                .toList()
        );
        
        // 3. 返回 Top K
        return rerankResults.stream()
            .sorted(Comparator.comparing(
                RerankResult::getScore).reversed())
            .limit(5)
            .map(result -> candidates.get(result.getIndex()))
            .toList();
    }
}
```

## Graph 框架实现原理

### 核心类结构

```
StateGraph
├── nodes: Map<String, Node>          # 节点映射
├── edges: Map<String, String>        # 边映射
├── conditionalEdges: Map<String, ConditionalEdge>
├── stateFactory: OverAllStateFactory  # 状态工厂
└── compile() → CompiledGraph         # 编译

CompiledGraph
├── nodeExecutors: Map<String, NodeExecutor>
├── edgeRouter: EdgeRouter
├── checkpointer: Checkpointer
└── execute()                         # 执行

OverAllState
├── data: Map<String, Object>         # 状态数据
├── strategies: Map<String, Strategy> # 更新策略
└── update(key, value)                # 状态更新
```

### 1. 状态管理实现

#### OverAllState 源码

```java
public class OverAllState implements Serializable {
    
    // 状态数据
    private final Map<String, Object> data = 
        new ConcurrentHashMap<>();
    
    // 更新策略
    private final Map<String, KeyStrategy> strategies = 
        new ConcurrentHashMap<>();
    
    // 注册键和策略
    public void registerKeyAndStrategy(
            String key, 
            KeyStrategy strategy) {
        strategies.put(key, strategy);
    }
    
    // 更新状态
    public void put(String key, Object value) {
        KeyStrategy strategy = strategies.get(key);
        if (strategy == null) {
            throw new IllegalArgumentException(
                "No strategy registered for key: " + key);
        }
        
        Object oldValue = data.get(key);
        Object newValue = strategy.merge(oldValue, value);
        data.put(key, newValue);
    }
    
    // 获取状态
    @SuppressWarnings("unchecked")
    public <T> T get(String key) {
        return (T) data.get(key);
    }
}
```

```java
// 替换策略
public class ReplaceStrategy implements KeyStrategy {
    @Override
    public Object merge(Object oldValue, Object newValue) {
        return newValue;  // 直接替换
    }
}

// 追加策略
public class AppendStrategy implements KeyStrategy {
    @Override
    public Object merge(Object oldValue, Object newValue) {
        if (oldValue == null) {
            return new ArrayList<>(List.of(newValue));
        }
        
        @SuppressWarnings("unchecked")
        List<Object> list = new ArrayList<>((List<Object>) oldValue);
        list.add(newValue);
        return list;
    }
}

// 合并策略
public class MergeStrategy implements KeyStrategy {
    @Override
    public Object merge(Object oldValue, Object newValue) {
        if (oldValue == null) {
            return new HashMap<>((Map<?, ?>) newValue);
        }
        
        @SuppressWarnings("unchecked")
        Map<Object, Object> result = 
            new HashMap<>((Map<Object, Object>) oldValue);
        result.putAll((Map<?, ?>) newValue);
        return result;
    }
}
```

### 2. 图编译过程

```java
public class StateGraph {
    
    public CompiledGraph compile(CompileConfig config) 
            throws GraphStateException {
        
        // 1. 图结构验证
        validate();
        
        // 2. 拓扑排序 (检测循环)
        List<String> topologicalOrder = topologicalSort();
        
        // 3. 构建执行计划
        ExecutionPlan plan = buildExecutionPlan();
        
        // 4. 创建节点执行器
        Map<String, NodeExecutor> executors = 
            createNodeExecutors();
        
        // 5. 构建已编译图
        return CompiledGraph.builder()
            .name(this.name)
            .stateFactory(this.stateFactory)
            .nodeExecutors(executors)
            .edgeRouter(new DefaultEdgeRouter(
                this.edges, 
                this.conditionalEdges
            ))
            .checkpointer(config.getCheckpointer())
            .lifecycleListeners(config.getLifecycleListeners())
            .build();
    }
    
    private void validate() throws GraphStateException {
        // 检查必须有 START 和 END
        if (!edges.containsKey(START)) {
            throw new GraphStateException("No edge from START");
        }
        
        // 检查所有边指向的节点都存在
        for (String target : edges.values()) {
            if (!END.equals(target) && !nodes.containsKey(target)) {
                throw new GraphStateException(
                    "Node not found: " + target);
            }
        }
        
        // 检查条件边
        for (ConditionalEdge edge : conditionalEdges.values()) {
            for (String target : edge.getMapping().values()) {
                if (!END.equals(target) && !nodes.containsKey(target)) {
                    throw new GraphStateException(
                        "Node not found in conditional edge: " + target);
                }
            }
        }
    }
}
```

### 3. 图执行引擎

```java
public class CompiledGraph {
    
    public Optional<OverAllState> invoke(Map<String, Object> input)
            throws GraphRunnerException {
        
        // 1. 初始化状态
        OverAllState state = stateFactory.create();
        input.forEach(state::put);
        
        // 2. 从 START 开始
        String currentNode = START;
        
        // 3. 执行循环
        while (!END.equals(currentNode)) {
            // 触发生命周期事件
            notifyBeforeNode(currentNode, state);
            
            // 执行节点
            NodeOutput output = executeNode(currentNode, state);
            
            // 更新状态
            state = output.getState();
            
            // 触发生命周期事件
            notifyAfterNode(currentNode, state);
            
            // 确定下一个节点
            currentNode = edgeRouter.route(currentNode, state);
            
            // 检查点保存
            if (checkpointer != null) {
                checkpointer.save(state);
            }
        }
        
        return Optional.of(state);
    }
    
    private NodeOutput executeNode(
            String nodeName, 
            OverAllState state) {
        
        NodeExecutor executor = nodeExecutors.get(nodeName);
        if (executor == null) {
            throw new IllegalStateException(
                "No executor for node: " + nodeName);
        }
        
        try {
            return executor.execute(state);
        } catch (Exception e) {
            throw new GraphRunnerException(
                "Node execution failed: " + nodeName, e);
        }
    }
}
```

```java
            }
            
            sink.complete();
        } catch (Exception e) {
            sink.error(e);
        }
    });
```

### 5. 并行节点执行

```java
public class ParallelExecutor {
    
    private final ExecutorService executor;
    
    public OverAllState executeParallel(
            List<String> nodeNames,
            OverAllState state) {
        
        // 1. 为每个节点创建副本状态
        List<OverAllState> stateCopies = nodeNames.stream()
            .map(name -> state.deepCopy())
            .toList();
        
        // 2. 并行执行
        List<CompletableFuture<NodeOutput>> futures = 
            IntStream.range(0, nodeNames.size())
                .mapToObj(i -> CompletableFuture.supplyAsync(
                    () -> executeNode(
                        nodeNames.get(i), 
                        stateCopies.get(i)
                    ),
                    executor
                ))
                .toList();
        
        // 3. 等待所有完成
        CompletableFuture.allOf(
            futures.toArray(new CompletableFuture[0])
        ).join();
        
        // 4. 合并结果
        List<OverAllState> results = futures.stream()
            .map(CompletableFuture::join)
            .map(NodeOutput::getState)
            .toList();
        
        return mergeStates(results);
    }
    
    private OverAllState mergeStates(List<OverAllState> states) {
        OverAllState merged = stateFactory.create();
        
        // 根据策略合并每个键的值
        for (String key : getAllKeys(states)) {
            KeyStrategy strategy = getStrategy(key);
            Object mergedValue = null;
            
            for (OverAllState state : states) {
                Object value = state.get(key);
                mergedValue = strategy.merge(mergedValue, value);
            }
            
            merged.put(key, mergedValue);
        }
        
        return merged;
    }
}
```

```java
public interface McpServiceDiscovery {
    // 根据服务名获取服务信息
    McpServerInfo getService(String serviceName);
}

@Slf4j
public class NacosMcpServiceDiscovery 
        implements McpServiceDiscovery {
    
    private final NamingService namingService;
    private final Map<String, McpServerInfo> cache;
    
    @Override
    public McpServerInfo getService(String serviceName) {
        // 1. 检查缓存
        McpServerInfo cached = cache.get(serviceName);
        if (cached != null && !isExpired(cached)) {
            return cached;
        }
        
        try {
            // 2. 从 Nacos 查询
            List<Instance> instances = namingService
                .selectInstances(serviceName, true);
            
            if (instances.isEmpty()) {
                log.warn("No healthy instances for: {}", serviceName);
                return null;
            }
            
            // 3. 选择实例 (负载均衡)
            Instance instance = selectInstance(instances);
            
            // 4. 构建服务信息
            McpServerInfo serverInfo = buildServerInfo(
                serviceName, 
                instance
            );
            
            // 5. 更新缓存
            cache.put(serviceName, serverInfo);
            
            return serverInfo;
            
        } catch (NacosException e) {
            log.error("Failed to get service from Nacos", e);
            return cached;  // 降级返回缓存
        }
    }
    
    private Instance selectInstance(List<Instance> instances) {
        // 简单的随机负载均衡
        int index = ThreadLocalRandom.current()
            .nextInt(instances.size());
        return instances.get(index);
    }
}
```

```java
public class McpServerVectorStore {
    
    private final VectorStore vectorStore;
    private final EmbeddingModel embeddingModel;
    private final Map<String, McpServerInfo> serverRegistry;
    
    public void addServer(McpServerInfo serverInfo) {
        // 1. 构建描述文本
        String description = buildDescription(serverInfo);
        
        // 2. 向量化
        List<Double> embedding = embeddingModel.embed(description);
        
        // 3. 存储
        Document doc = Document.builder()
            .id(serverInfo.getName())
            .text(description)
            .metadata("name", serverInfo.getName())
            .metadata("endpoint", serverInfo.getEndpoint())
            .build();
        
        vectorStore.add(List.of(doc));
        
        // 4. 注册
        serverRegistry.put(serverInfo.getName(), serverInfo);
    }
    
    public List<McpServerInfo> search(
            String query, 
            int topK) {
        
        // 1. 语义搜索
        List<Document> results = vectorStore.similaritySearch(
            SearchRequest.query(query)
                .withTopK(topK)
                .withSimilarityThreshold(0.6)
        );
        
        // 2. 转换为服务信息
        return results.stream()
            .map(doc -> serverRegistry.get(
                doc.getMetadata().get("name")
            ))
            .filter(Objects::nonNull)
            .toList();
    }
    
    private String buildDescription(McpServerInfo serverInfo) {
        StringBuilder sb = new StringBuilder();
        sb.append("服务名称: ").append(serverInfo.getName())
          .append("\n");
        sb.append("描述: ").append(serverInfo.getDescription())
          .append("\n");
        sb.append("工具列表:\n");
        
        for (ToolInfo tool : serverInfo.getTools()) {
            sb.append("- ").append(tool.getName())
              .append(": ").append(tool.getDescription())
              .append("\n");
        }
        
        return sb.toString();
    }
}
```

```java
        // 2. 向量搜索
        List<McpServerInfo> servers = 
            mcpServerVectorStore.search(
                searchQuery, 
                limit != null ? limit : 5
            );
        
        // 3. 构建响应
        return McpServerSearchResponse.builder()
            .servers(servers)
            .query(searchQuery)
            .count(servers.size())
            .build();
    }
    
    @Tool(description = "使用工具")
    public McpToolExecutionResponse useTool(
            @Description("服务名称") String serverName,
            @Description("工具名称") String toolName,
            @Description("工具参数JSON") String toolArguments) {
        
        try {
            // 1. 获取服务信息
            McpServerInfo serverInfo = 
                mcpServiceDiscovery.getService(serverName);
            
            if (serverInfo == null) {
                return McpToolExecutionResponse.error(
                    "Service not found: " + serverName
                );
            }
            
            // 2. 调用工具
            Object result = mcpProxyService.callTool(
                serverInfo.getEndpoint(),
                toolName,
                toolArguments
            );
            
            // 3. 返回结果
            return McpToolExecutionResponse.success(result);
            
        } catch (Exception e) {
            log.error("Tool execution failed", e);
            return McpToolExecutionResponse.error(
                e.getMessage()
            );
        }
    }
}
```

```java
public class JsonRpcA2aRequestHandler {
    
    private final JSONRPCHandler jsonRpcHandler;
    
    public Object onHandler(String body, Headers headers) {
        // 1. 判断是否流式请求
        boolean streaming = isStreamingRequest(body);
        
        // 2. 处理请求
        if (streaming) {
            return handleStreamRequest(body);
        } else {
            return handleNonStreamRequest(body);
        }
    }
    
    private Flux<?> handleStreamRequest(String body) {
        return Flux.create(sink -> {
            try {
                // 1. 解析请求
                JSONRPCRequest request = parseRequest(body);
                
                // 2. 创建事件队列
                EventQueue eventQueue = new EventQueue();
                
                // 3. 执行 Agent
                executor.execute(
                    createContext(request), 
                    eventQueue
                );
                
                // 4. 流式发送事件
                eventQueue.subscribe(event -> {
                    JSONRPCResponse response = 
                        createResponse(request.getId(), event);
                    sink.next(response);
                    
                    if (event.isFinal()) {
                        sink.complete();
                    }
                });
                
            } catch (Exception e) {
                sink.error(e);
            }
        });
    }
}
```

```java
public class GraphAgentExecutor implements AgentExecutor {
    
    private final Agent agent;
    
    @Override
    public void execute(
            RequestContext context, 
            EventQueue eventQueue) {
        
        try {
            // 1. 提取消息
            String input = extractInput(context);
            
            // 2. 准备输入
            Map<String, Object> input = Map.of(
                "messages", 
                List.of(new UserMessage(input))
            );
            
            // 3. 流式执行 Agent
            if (isStreamRequest(context)) {
                agent.stream(input)
                    .subscribe(
                        output -> sendOutput(output, eventQueue),
                        error -> sendError(error, eventQueue),
                        () -> sendComplete(eventQueue)
                    );
            } else {
                Optional<OverAllState> result = agent.invoke(input);
                sendResult(result, eventQueue);
            }
            
        } catch (Exception e) {
            eventQueue.enqueueEvent(
                A2A.toAgentMessage("Error: " + e.getMessage())
            );
        }
    }
    
    private void sendOutput(
            NodeOutput output, 
            EventQueue eventQueue) {
        
        // 构建事件
        TaskStatusUpdateEvent event = 
            TaskStatusUpdateEvent.builder()
                .taskId(output.getTaskId())
                .status("in_progress")
                .message(extractMessage(output))
                .build();
        
        eventQueue.enqueueEvent(event);
    }
}
```

```java
public class ChatModelInputObservationHandler

    private final AttributeKey<String> inputMessagesKey;

    public void onStop(ChatModelObservationContext context) {

            .retrieveOtelSpan(tracingContext);

            // 3. 序列化输入消息

            // 4. 设置 Span 属性
            otelSpan.setAttribute(inputMessagesKey, inputMessages);
        }
    }
    
    private String serializeInputMessages(
            ChatModelObservationContext context) {
        
        List<ChatMessage> messages = context.getRequest()
            .getInstructions()
            .stream()
            .map(this::convertMessage)
            .toList();
        
        try {
            return objectMapper.writeValueAsString(messages);
        } catch (JsonProcessingException e) {
            log.warn("Failed to serialize messages", e);
            return "[]";
        }
    }
}
```
